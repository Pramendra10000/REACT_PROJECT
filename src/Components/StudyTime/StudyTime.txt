
================= IF WE NEED TO USE CONSOLE FOR ADDING DATA IN JAVA SO WE CAN USE BELOW =====================

(async function () {

  const data = {
    "Core Java Basics": [
      { q: "What is JVM?", a: "Java Virtual Machine" },
      { q: "What is JDK?", a: "Java Development Kit" }
    ]
  };

  const wait = (ms) => new Promise(res => setTimeout(res, ms));

  function getAddBtn() {
    return [...document.querySelectorAll("button")]
      .find(btn => btn.textContent.includes("Add"));
  }

  async function waitForNewCard(prevCount) {
    for (let i = 0; i < 20; i++) {
      const cards = document.querySelectorAll(".addqa-card");
      if (cards.length > prevCount) return cards;
      await wait(300);
    }
    throw new Error("❌ Card not appearing");
  }

  function setReactValue(element, value) {
    const setter = Object.getOwnPropertyDescriptor(
      element.__proto__,
      "value"
    ).set;

    setter.call(element, value);
    element.dispatchEvent(new Event("input", { bubbles: true }));
  }

  let cards = document.querySelectorAll(".addqa-card");

  // ✅ Use FIRST existing card (NO extra empty)
  let isFirstEntry = true;

  for (const topic in data) {
    for (const item of data[topic]) {

      if (!isFirstEntry) {
        const addBtn = getAddBtn();
        const prevCount = document.querySelectorAll(".addqa-card").length;

        addBtn.click();

        cards = await waitForNewCard(prevCount);
      } else {
        cards = document.querySelectorAll(".addqa-card");
      }

      const currentCard = isFirstEntry
        ? cards[0]
        : cards[cards.length - 1];

      const inputs = currentCard.querySelectorAll("input");
      const textarea = currentCard.querySelector("textarea");

      setReactValue(inputs[0], topic);
      setReactValue(inputs[1], item.q);
      setReactValue(textarea, item.a);

      isFirstEntry = false;

      await wait(300);
    }
  }

  console.log("✅ PERFECT: No empty card, all filled!");

})();
